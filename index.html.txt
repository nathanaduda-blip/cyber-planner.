  <script>
    const days = ['mon', 'tue', 'wed', 'thu', 'fri', 'sat', 'sun'];
    const timeSlots = [];
    let swRegistration = null;

    if ('serviceWorker' in navigator) {
      window.addEventListener('load', () => {
        navigator.serviceWorker.register('./sw.js')
          .then(reg => { swRegistration = reg; })
          .catch(err => console.error('Service Worker registration failed: ', err));
      });
    }

    function requestNotificationPermission() {
      if ('Notification' in window) {
        Notification.requestPermission().then(permission => {
          if (permission === 'granted') {
            alert('Notifications enabled successfully!');
          } else {
            alert('Notification permission denied.');
          }
        });
      }
    }

    setInterval(() => {
      if (Notification.permission !== 'granted') return;

      const now = new Date();
      const currentDay = days[(now.getDay() + 6) % 7];
      const currentTime = `${String(now.getHours()).padStart(2, '0')}:${String(now.getMinutes()).padStart(2, '0')}`;
      
      if (now.getSeconds() === 0) {
        const saved = localStorage.getItem('cyberGothicSchedule');
        if (!saved) return;

        const events = JSON.parse(saved);
        events.forEach(ev => {
          const startTimeString = timeSlots[ev.startIndex];
          if (ev.day === currentDay && startTimeString === currentTime) {
            triggerNotification(`Cyber Alert: ${ev.title}`, `Scheduled event starting right now (${currentTime})!`);
          }
        });
      }
    }, 1000);

    function triggerNotification(title, body) {
      if (swRegistration && swRegistration.active) {
        swRegistration.active.postMessage({ type: 'TRIGGER_NOTIFICATION', title, body });
      } else {
        new Notification(title, { body: body });
      }
    }

    function toggleTheme() {
      const body = document.body;
      const currentTheme = body.getAttribute('data-theme');
      const newTheme = currentTheme === 'dark' ? 'light' : 'dark';
      body.setAttribute('data-theme', newTheme);
      document.getElementById('themeLabel').textContent = newTheme.toUpperCase();
      localStorage.setItem('cyberTheme', newTheme);
    }

    function loadTheme() {
      const savedTheme = localStorage.getItem('cyberTheme') || 'light';
      document.body.setAttribute('data-theme', savedTheme);
      document.getElementById('themeLabel').textContent = savedTheme.toUpperCase();
    }

    function generateTable() {
      const tbody = document.getElementById('scheduleBody');
      const startHour = 6;
      const endHour = 22;

      for (let hour = startHour; hour < endHour; hour++) {
        for (let min = 0; min < 60; min += 15) {
          const timeString = `${String(hour).padStart(2, '0')}:${String(min).padStart(2, '0')}`;
          timeSlots.push(timeString);

          const tr = document.createElement('tr');
          const timeTd = document.createElement('td');
          timeTd.className = 'time-col';
          timeTd.textContent = timeString;
          tr.appendChild(timeTd);

          days.forEach(day => {
            const slotTd = document.createElement('td');
            slotTd.className = 'slot';
            slotTd.dataset.day = day;
            slotTd.dataset.time = timeString;
            slotTd.onclick = () => openModal(day, timeString);
            tr.appendChild(slotTd);
          });

          tbody.appendChild(tr);
        }
      }
      populateTimeSelects();
    }

    function populateTimeSelects() {
      const startSelect = document.getElementById('startTime');
      const endSelect = document.getElementById('endTime');

      timeSlots.forEach(t => {
        startSelect.innerHTML += `<option value="${t}">${t}</option>`;
        endSelect.innerHTML += `<option value="${t}">${t}</option>`;
      });
    }

    function openModal(day = 'mon', startTime = '08:00') {
      document.getElementById('eventDay').value = day;
      document.getElementById('startTime').value = startTime;
      
      const startIdx = timeSlots.indexOf(startTime);
      const endIdx = Math.min(startIdx + 2, timeSlots.length - 1);
      document.getElementById('endTime').value = timeSlots[endIdx];
      document.getElementById('modalOverlay').style.display = 'flex';
    }

    function closeModal() {
      document.getElementById('modalOverlay').style.display = 'none';
      document.getElementById('eventTitle').value = '';
    }

    function submitEvent() {
      const title = document.getElementById('eventTitle').value.trim();
      const day = document.getElementById('eventDay').value;
      const startTime = document.getElementById('startTime').value;
      const endTime = document.getElementById('endTime').value;
      const color = document.getElementById('eventColor').value;

      const startIdx = timeSlots.indexOf(startTime);
      const endIdx = timeSlots.indexOf(endTime);

      if (!title) { alert('Please enter an activity title.'); return; }
      if (endIdx <= startIdx) { alert('End time must be after start time.'); return; }

      const allDaySlots = Array.from(document.querySelectorAll(`td.slot[data-day="${day}"]`));
      const targetSlots = allDaySlots.slice(startIdx, endIdx);

      const hasOverlap = targetSlots.some(s => s.style.display === 'none' || s.children.length > 0);
      if (hasOverlap) { alert('This time slot overlaps with an existing activity.'); return; }

      createEventBlock(targetSlots, title, color);
      closeModal();
    }

    function createEventBlock(slots, title, color) {
      const spanCount = slots.length;
      const topSlot = slots[0];

      for (let i = 1; i < slots.length; i++) slots[i].style.display = 'none';

      topSlot.rowSpan = spanCount;
      const textColor = getContrastColor(color);
      topSlot.innerHTML = `<div class="event-block" style="background-color: ${color}; color: ${textColor};">${title}</div>`;

      const block = topSlot.querySelector('.event-block');
      block.onclick = (e) => {
        e.stopPropagation();
        if (confirm(`Delete "${title}"?`)) {
          deleteEventBlock(topSlot, slots);
        }
      };
    }

    function deleteEventBlock(topSlot, allSlots) {
      topSlot.innerHTML = '';
      topSlot.removeAttribute('rowSpan');
      allSlots.forEach(s => s.style.display = '');
    }

    function getContrastColor(hexcolor){
      hexcolor = hexcolor.replace("#", "");
      var r = parseInt(hexcolor.substr(0,2),16);
      var g = parseInt(hexcolor.substr(2,2),16);
      var b = parseInt(hexcolor.substr(4,2),16);
      var yiq = ((r*299)+(g*587)+(b*114))/1000;
      return (yiq >= 128) ? '#000000' : '#ffffff';
    }

    function saveSchedule() {
      const events = [];
      const topSlots = document.querySelectorAll('.slot[rowspan]');

      topSlots.forEach(slot => {
        const spanCount = parseInt(slot.rowSpan);
        const day = slot.dataset.day;
        const allDaySlots = Array.from(document.querySelectorAll(`td.slot[data-day="${day}"]`));
        const startIndex = allDaySlots.indexOf(slot);
        
        events.push({
          day: day,
          startIndex: startIndex,
          spanCount: spanCount,
          title: slot.querySelector('.event-block').innerText,
          color: slot.querySelector('.event-block').style.backgroundColor
        });
      });

      localStorage.setItem('cyberGothicSchedule', JSON.stringify(events));
      
      const status = document.getElementById('status');
      status.classList.add('show');
      setTimeout(() => status.classList.remove('show'), 2000);
    }

    function loadSchedule() {
      const saved = localStorage.getItem('cyberGothicSchedule');
      if (!saved) return;

      const events = JSON.parse(saved);
      events.forEach(ev => {
        const allDaySlots = Array.from(document.querySelectorAll(`td.slot[data-day="${ev.day}"]`));
        const targetSlots = allDaySlots.slice(ev.startIndex, ev.startIndex + ev.spanCount);
        if (targetSlots.length > 0) {
          createEventBlock(targetSlots, ev.title, ev.color);
        }
      });
    }

    function clearSchedule() {
      if (confirm('Are you sure you want to clear your schedule?')) {
        localStorage.removeItem('cyberGothicSchedule');
        location.reload();
      }
    }

    window.onload = () => {
      loadTheme();
      generateTable();
      loadSchedule();
    };
  </script>
